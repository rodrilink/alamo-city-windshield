Create `src/components/home/HeroSection.tsx` as a Client Component implementing the hero (per D-01, D-02, D-03, D-04, D-05):

```tsx
'use client'

import { type RefObject } from 'react'
import Image from 'next/image'
import { motion } from 'motion/react'
import { ChevronDown } from 'lucide-react'
import { Button } from '@/components/ui/button'

interface HeroSectionProps {
  scrollRef: RefObject<HTMLDivElement | null>
}

export function HeroSection({ scrollRef }: HeroSectionProps) {
  function scrollToEstimate() {
    const container = scrollRef.current
    if (!container) return
    // Scroll to the second snap section (one viewport height down)
    container.scrollTo({ top: container.clientHeight, behavior: 'smooth' })
  }

  return (
    <section className="snap-start snap-always h-dvh relative overflow-hidden flex items-center justify-center">
      {/* Background photo — Unsplash auto glass / windshield image */}
      <Image
        src="https://images.unsplash.com/photo-1558618666-fcd25c85cd64?w=1920&q=80"
        alt="Auto glass repair technician working on a windshield"
        fill
        className="object-cover"
        priority
      />
      {/* Dark gradient overlay at ~55% opacity (per D-03: 50-60%) */}
      <div className="absolute inset-0 bg-black/55" />

      {/* Content — centered, above overlay */}
      <div className="relative z-10 flex flex-col items-center text-center px-4 max-w-4xl">
        {/* Headline with entrance animation (per D-01, D-04) */}
        <motion.h1
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.7, ease: 'easeOut', delay: 0.2 }}
          className="font-display text-4xl font-bold text-white md:text-5xl lg:text-6xl"
        >
          San Antonio&apos;s Trusted Windshield Repair &amp; Replacement
        </motion.h1>

        {/* Subheadline */}
        <motion.p
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.7, ease: 'easeOut', delay: 0.4 }}
          className="mt-4 text-lg text-white/80 max-w-2xl md:text-xl"
        >
          Expert mobile and in-shop service across San Antonio, TX.
          Get a free estimate in seconds.
        </motion.p>

        {/* CTA button (per D-02): brand red bg, white text, scrolls to section 2 */}
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.7, ease: 'easeOut', delay: 0.6 }}
        >
          <Button
            size="lg"
            className="mt-8 text-base px-8 py-3 h-auto"
            onClick={scrollToEstimate}
          >
            Get Free Estimate
          </Button>
        </motion.div>
      </div>

      {/* Bouncing down-arrow (per D-05) — Tailwind animate-bounce, no extra JS */}
      <div className="absolute bottom-8 left-1/2 -translate-x-1/2 z-10 animate-bounce text-white/80">
        <ChevronDown className="h-8 w-8" />
      </div>
    </section>
  )
}
```

Key implementation details:
- **D-01**: Exact headline text "San Antonio's Trusted Windshield Repair & Replacement"
- **D-02**: "Get Free Estimate" button with `onClick={scrollToEstimate}` that calls `scrollRef.current.scrollTo()`. Brand red via default Button variant (bg-primary = brand red). White text via primary-foreground.
- **D-03**: Unsplash auto glass photo with `fill` + `object-cover` + `<div className="absolute inset-0 bg-black/55">` overlay. Parent has `relative overflow-hidden`.
- **D-04**: Hero elements use `initial={{ opacity: 0, y: 20 }}` and `animate={{ opacity: 1, y: 0 }}` with staggered delays (0.2, 0.4, 0.6s). This is mount-based animation (NOT whileInView) since the hero is visible on page load.
- **D-05**: `animate-bounce` on a ChevronDown icon at `bottom-8`. Tailwind utility, zero extra JS.
- `priority` on the hero Image for immediate load (LCP optimization)
- No `whileInView` on hero — hero animates on mount. `whileInView` is for sections 2 and 3 (Plans 02 and 03).

**Important: The Button component uses `@base-ui/react` which does NOT support `asChild`.** Use `onClick` handler for scroll behavior, not `asChild` with an anchor. For link navigation in other buttons, wrap `Button` around the text content or use a different pattern — confirmed in button.tsx source.

Choose an Unsplash photo URL. The recommended URL `photo-1558618666-fcd25c85cd64` is an auto glass technician image. If this URL is not suitable at implementation time (404 or wrong subject), substitute any Unsplash photo matching: windshield, auto glass, car window repair. Append `?w=1920&q=80` for optimized resolution.
